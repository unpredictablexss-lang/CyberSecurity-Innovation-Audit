  --detector DETECTOR   Explicit first-party detector ID (e.g.
                        ohscanner.interface.ipc); repeatable. Not an import
                        path.

                        Acknowledge that 'ohscanner.example.*' is a non-
                        security demonstration
  --json-out JSON_OUT
  --md-out MD_OUT
